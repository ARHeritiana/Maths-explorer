\section{Proportions and percentage change}

\fichehead{Tue~2~Feb.~2027}{Proportions and percentages; proportion of a proportion}
\begin{fiche}
\qitem{Warm-up · Quick}{PRO}{Calculate $15\,\%$ of $240$.}{$0.15\times240=36$.}{}
\qitem{Core · Method}{PRO}{$30\,\%$ of the pupils are boarders and $40\,\%$ of the boarders do dance. What proportion of the pupils are boarders who do dance?}{$0.3\times0.4=0.12=12\,\%$: proportion of a proportion (we multiply).}{}
\qitem{Core + · Problem}{PRO}{A school has $800$ pupils, of whom $55\,\%$ are girls; $20\,\%$ of the girls study Latin. How many girls study Latin?}{$800\times0.55=440$ girls; $440\times0.20=88$.}{}
\qitem{Review · Quick}{QP}{$3$ exercise books cost $4.50$ €. Price of $7$ books?}{$1.50\times7=10.50$ €.}{}
\qitem{Thread · True/false}{EQ}{True or false? Justify. The number $-2$ is a solution of the equation $3x+1=-5$.}{True. $3\times(-2)+1=-6+1=-5$.}{}
\end{fiche}

\fichehead{Thu~4~Feb.~2027}{Absolute change, multiplier, rate of change}
\begin{fiche}
\qitem{Warm-up · Quick}{PRO}{A price goes from $80$ Ar to $92$ Ar. What is the absolute change?}{$92-80=12$ Ar.}{}
\qitem{Core · Method}{PRO}{For the same change, give the rate of change and the multiplier.}{$t=\dfrac{12}{80}=0.15=15\,\%$; $CM=1+t=1.15$.}{}
\qitem{Core + · MCQ}{PRO}{A decrease of $20\,\%$ corresponds to the multiplier:\choix{$0.2$}{$-0.8$}{$1.2$}{$0.8$}}{D.\ $0.8$}{$CM=1+t=1-0.20$.}
\qitem{Review · Quick}{STA}{Temperatures $-2,\,0,\,3,\,5,\,9$ (in °C). Mean and range?}{Mean $3$ °C; range $9-(-2)=11$ °C.}{}
\qitem{Thread · Quick}{SOL}{Calculate the volume and the total surface area of a cube with edge $3$ cm.}{$V=3^3=27$ cm$^3$; $A=6\times3^2=54$ cm$^2$.}{}
\end{fiche}

\fichehead{Mon~8~Feb.~2027}{Successive percentage changes}
\begin{fiche}
\qitem{Warm-up · Quick}{CAL}{Calculate $1.2\times0.9$.}{$1.08$.}{}
\qitem{Core · Method}{PRO}{An increase of $20\,\%$ is followed by a decrease of $10\,\%$. What is the overall rate of change?}{$1.2\times0.9=1.08$: overall increase of $8\,\%$.}{Rates are not added; it is the multipliers that are multiplied.}
\qitem{Core + · True/false}{PRO}{True or false? Justify. An increase of $50\,\%$ followed by a decrease of $50\,\%$ returns to the initial value.}{False. $1.5\times0.5=0.75$: overall decrease of $25\,\%$.}{}
\qitem{Review · Quick}{SUB}{Calculate $x^2-x$ for $x=\dfrac12$.}{$-\dfrac{1}{4}$.}{}
\qitem{Thread · Write-up}{DEM}{$ABCD$ is a parallelogram such that $AC=BD$. What type of quadrilateral is it? Justify in one sentence.}{Rectangle: a parallelogram whose diagonals have equal length is a rectangle.}{}
\end{fiche}

\fichehead{Tue~9~Feb.~2027}{Reverse percentage change; proportion and change}
\begin{fiche}
\qitem{Warm-up · Quick}{CAL}{Calculate $\dfrac{1}{1.25}$.}{$0.8$.}{}
\qitem{Core · Method}{PRO}{After an increase of $25\,\%$, what decrease brings back the initial value?}{$CM_{\text{rev}}=\dfrac{1}{1.25}=0.8$: decrease of $20\,\%$ (not $25\,\%$).}{Pitfall: believing that an increase of $25\,\%$ is cancelled by a decrease of $25\,\%$.}
\qitem{Core + · Interpret}{PRO}{Out of $250$ candidates, $150$ are successful; last year there were $200$ candidates. Calculate the proportion of successful candidates, then the rate of change of the number of candidates.}{Proportion $\dfrac{150}{250}=60\,\%$ (part over whole); change $\dfrac{250-200}{200}=25\,\%$ (relative change).}{}

$f(x)$ & 4 & 1 & 0 & 1 \\ \hline
\end{tabular}\par Image of $2$? Preimages of $1$?}{$f(2)=0$; preimages of $1$: $1$ and $3$.}{}
\qitem{Thread · MCQ}{REP}{On the globe, two points on the same parallel have:\choix{the same latitude}{the same longitude}{the same distance from Greenwich}{the same altitude}}{A.\ the same latitude}{Latitude: north/south angle from the equator; longitude: east/west angle from the Greenwich meridian.}
