\medskip
\emph{Pour chacune des affirmations suivantes, indiquer si elle est vraie ou fausse. Chaque réponse doit être justifiée. Une réponse non justifiée ne rapporte aucun point.
}
\begin{enumerate}
	\item On considère la fonction $f$ définie sur $\R$ par :\quad $f(x)=5 x \e^{-x}$.\\
	On note $C_{f}$ la courbe représentative de $f$ dans un repère orthonormé.

	\smallskip

	\textbf{Affirmation 1 :}\\
	L'axe des abscisses est une asymptote horizontale à la courbe $C_{f}$.

	\smallskip

	\textbf{Affirmation 2 :}\\
	La fonction $f$ est solution sur $\R$ de l'équation différentielle $(E)$ : $y'+y=5 \e^{-x}$.
	\smallskip

	\item On considère les suites $\left(u_{n}\right)$, $\left(v_{n}\right)$ et $\left(w_{n}\right)$, telles que, pour tout entier naturel $n$ :
	$$
	u_{n} \leqslant v_{n} \leqslant w_{n}.
	$$
	De plus, la suite $\left(u_{n}\right)$ converge vers $-1$ et la suite $\left(w_{n}\right)$ converge vers 1 .

	\smallskip
	\textbf{Affirmation 3 :}\\
	La suite $\left(v_{n}\right)$ converge vers un nombre réel $\ell$ appartenant à l'intervalle $[-1 ; 1]$.


	\smallskip
	On suppose de plus que la suite $\left(u_{n}\right)$ est croissante et que la suite $\left(w_{n}\right)$ est décroissante.

	\textbf{Affirmation 4 :}\\
	Pour tout entier naturel $n$, on a alors : \quad $u_{0} \leqslant v_{n} \leqslant w_{0}$.

\end{enumerate}
