\emph{Pour chaque affirmation, on indique si elle est vraie ou fausse en justifiant. Les deux
questions sont indépendantes.}

\begin{enumerate}
	\item \textbf{Affirmation vraie.}

	$f$ est un produit $uv$ avec $u(x) = x$, $u'(x) = 1$ et $v(x) = \e^x$, $v'(x) = \e^x$.
	Pour tout $x$ de $[0~;~+\infty[$ :
	\[f'(x) = 1 \times \e^x + x \times \e^x = (1+x)\e^x.\]
	Pour tout $x \geqslant 0$, on a $1 + x \geqslant 1 > 0$ et $\e^x > 0$, donc $f'(x) > 0$ :
	la fonction $f$ est croissante sur $[0~;~+\infty[$.

	\item \textbf{Affirmation vraie.}

	Dans une suite arithmétique de raison $r$ : $u_n = u_p + (n-p) \times r$.
	
	Ainsi : $u_6 = u_2 + 4r$.
	\[u_6 = u_2 + 4r \iff 51 = 50 + 4r \iff 4r = 1 \iff r = \dfrac{1}{4} = 0,25.\]
	D'où :
	\[u_{26} = u_2 + 24r = 50 + 24 \times 0,25 = 50 + 6 = 56.\]
\end{enumerate}
