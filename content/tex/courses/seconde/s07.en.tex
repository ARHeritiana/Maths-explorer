\section{Frequencies, relative frequencies and mean}
\begin{keypoint}[Definitions]
A data set describes a characteristic over a population. If the value $x_i$ appears $n_i$ times, $n_i$ is its frequency count. The nonzero total count is $N=\sum_{i=1}^{p}n_i$; its relative frequency is $f_i=n_i/N$. The mean is
\[\overline x=\frac{\sum_{i=1}^{p}n_ix_i}{N}=\sum_{i=1}^{p}f_ix_i.\]
The range is $x_{\max}-x_{\min}$.
\end{keypoint}
\begin{example}[Shoe sizes -- booklet data]We study the shoe sizes of $30$ students:
\begin{center}\small\begin{tabular}{rrrrrrrrrrrrrrr}
36&42&40&41&41&38&39&43&42&42&42&40&36&40&39\\
41&40&38&44&42&40&42&43&38&43&36&41&41&44&43
\end{tabular}\end{center}
Complete the counts, calculate the relative frequencies, the mean and the range.\end{example}
\begin{center}\mxarraystretch{1.6}\begin{tabular}{c|rrrrrrrrr}
Size&36&37&38&39&40&41&42&43&44\\\hline
Count&\mxblank{3}&\mxblank{0}&\mxblank{3}&\mxblank{2}&\mxblank{5}&\mxblank{5}&\mxblank{6}&\mxblank{4}&\mxblank{2}
\end{tabular}\end{center}
\mxanswerspace[6]{Counts: $3,0,3,2,5,5,6,4,2$, with sum $30$. The relative frequencies are these counts divided by $30$, that is as percentages $10$; $0$; $10$; about $6.67$; $16.67$; $16.67$; $20$; $13.33$; $6.67$.
$\sum n_ix_i=1217$, so $\overline x=1217/30\approx40.57$. The range is $8$. The mean is not necessarily a value of the data set. The sum of the rounded percentages may differ slightly from $100$.}
\mxsection{Median and cumulative counts}
\begin{keypoint}[Definition and convention]
A median is a number such that at least half of the values are less than or equal to it, and at least half are greater than or equal to it.
In the ordered data set of $N$ values, if $N$ is odd, take the value of rank $(N+1)/2$; if $N$ is even, choose the mean of the values of ranks $N/2$ and $N/2+1$.
\end{keypoint}
\begin{pitfall}With repeated values, more than $50\,\%$ of the values can be less than or equal to the median.\end{pitfall}
\begin{example}Complete the ascending cumulative counts (ACC) of the shoe sizes and determine the median.\end{example}
\begin{center}\mxarraystretch{1.7}\begin{tabular}{c|rrrrrrrrr}
Size&36&37&38&39&40&41&42&43&44\\\hline
Count&3&0&3&2&5&5&6&4&2\\
ACC&\mxblank{3}&\mxblank{3}&\mxblank{6}&\mxblank{8}&\mxblank{13}&\mxblank{18}&\mxblank{24}&\mxblank{28}&\mxblank{30}
\end{tabular}\end{center}
\mxanswerspace[4]{The central ranks are $15$ and $16$. The ACC goes from $13$ for $40$ to $18$ for $41$: both central values equal $41$, so $M_e=41$. At least half of the sizes are less than or equal to $41$, and at least half are greater than or equal to $41$.}
\begin{example}Calculate the medians of $1;2;2;8;9$ and of $1;2;8;9$.\end{example}
\mxanswerspace[3.2]{The third value of the first data set is $2$. In the second, $(2+8)/2=5$; the chosen median is not a value of the data set.}
\mxsection{Quartiles and box plot}
\begin{keypoint}[Definitions]
The first quartile $Q_1$ is the value of rank $\lceil N/4\rceil$: the smallest value of the data set such that at least $25\,\%$ of the values are less than or equal to it. $Q_3$ is the value of rank $\lceil3N/4\rceil$.
The interquartile range is $Q_3-Q_1$. The box plot shows the minimum, $Q_1$, the median, $Q_3$ and the maximum.
\end{keypoint}
\begin{example}Determine $Q_1$, $Q_3$ and the interquartile range of the shoe sizes, then draw the box plot.\end{example}
\mxanswerspace[3.2]{$N/4=7.5$: rank $8$, so $Q_1=39$. $3N/4=22.5$: rank $23$, so $Q_3=42$. Interquartile range $3$. The plot uses $(36;39;41;42;44)$.}

\mxsection{Adding or removing a value}
\begin{keypoint}[Properties]
Adding $a$ to a data set of size $N$ gives $\overline x'=(N\overline x+a)/(N+1)$.
Removing one occurrence of $a$, for $N>1$, gives $\overline x'=(N\overline x-a)/(N-1)$. Use the exact sum, with no intermediate rounding. Re-examine the median in the new ordered data set.
\end{keypoint}
\begin{example}In the shoe-size data, the sum is $1217$ and the median $41$.
\begin{enumerate}\item Add a size $36$ and calculate the new statistics.\item Starting again from the initial data, remove a size $44$ and repeat.\end{enumerate}\end{example}
\mxanswerspace[5]{1. Sum $1253$, count $31$: $\overline x'=1253/31\approx40.42$. The central rank $16$ still gives $41$. 2. Sum $1173$, count $29$: $\overline x''=1173/29\approx40.45$. The central rank $15$ gives $41$. The mean decreases in both cases; the median is unchanged. The situations are independent.}
\begin{example}Justify that adding the mean does not change the mean. Can the range decrease when a value is added?\end{example}
\mxanswerspace[4]{If $a=\overline x$, $(N\overline x+\overline x)/(N+1)=\overline x$. When adding, the minimum can only stay the same or decrease, the maximum stay the same or increase: the range does not decrease. When removing, it does not increase.}
\mxsection{Variance and standard deviation}
\begin{keypoint}[Definitions]
$V=\dfrac{\sum n_i(x_i-\overline x)^2}{N}$ is the variance; $\sigma=\sqrt V$ is the standard deviation. It measures the spread around the mean and has the same unit as the data. One can also compute $V=\dfrac{\sum n_ix_i^2}{N}-\overline x^{\,2}$.
\end{keypoint}
\begin{example}Calculate the variance and the standard deviation of the shoe sizes.\end{example}
\mxanswerspace[4]{$\sum n_ix_i^2=49519$, so
\[V=\frac{49519}{30}-\left(\frac{1217}{30}\right)^2=\frac{4481}{900}\approx4.979.\]
$\sigma=\sqrt{4481}/30\approx2.23$. The standard deviation is not the maximum distance from the mean.}
\begin{example}Compare $A:8;10;12$ and $B:0;10;20$ using the mean--standard deviation pair and the median--interquartile range pair.\end{example}
\mxanswerspace[5]{The means and medians are all $10$. $V_A=8/3$, $\sigma_A\approx1.63$; $V_B=200/3$, $\sigma_B\approx8.16$. Quartiles: $8,12$ for $A$; $0,20$ for $B$. Interquartile ranges $4$ and $20$. $B$ is much more spread out despite identical centres.}
\begin{teachernote}The descriptive variance divides by $N$, with no $N-1$ correction. Also check the quartile convention of software.\end{teachernote}
\mxsection{Linearity of the mean}
\begin{keypoint}[Property]
Replacing each $x_i$ by $kx_i+c$ gives a mean $k\overline x+c$, a standard deviation $|k|\sigma$ and a variance $k^2V$. Adding a constant changes neither variance nor standard deviation.
\end{keypoint}
\begin{example}Initial mean $12/20$, standard deviation $3$. Calculate the new statistics for three independent modifications, with no capping of marks: add $1.5$; multiply by $0.8$; multiply by $1.1$ then add $0.5$.\end{example}
\mxanswerspace[4]{Means: $13.5$, $9.6$, $13.7$. Standard deviations: $3$, $2.4$, $3.3$. Multiplying the marks by $1.1$ raises them; this does not mean making the assessment harder. With a cap at $20$, the transformation would no longer be affine on all the data.}
\begin{lstlisting}
from math import sqrt

def moyenne_ecart_type(valeurs):
    n = len(valeurs)
    if n == 0:
        raise ValueError("liste vide")
    m = sum(valeurs)/n
    v = sum((x-m)**2 for x in valeurs)/n
    return m, sqrt(v)
\end{lstlisting}
\begin{example}Explain the program; predict its result for \texttt{[8, 10, 12]}.\end{example}
\mxanswerspace[3.2]{It checks that the list is nonempty, calculates the mean, the mean of the squared deviations, then its root. Result: $(10;\sqrt{8/3})\approx(10;1.633)$.}
\mxsection{Continuous data grouped into classes}
\begin{keypoint}[Definitions and loss of information]
A continuous quantitative variable can take every value in an interval. Here, the observations are grouped into classes of equal width; the centre of $[a;b[$ is $(a+b)/2$.
If the class means $m_i$ are known, the exact overall mean is $\sum n_im_i/N$. Otherwise, replacing them by the centres gives an estimate justified by a uniform model within each class.
\end{keypoint}
\begin{example}[Heights -- booklet data]Complete, estimate the mean from the centres and determine the median class.\end{example}
\begin{center}\mxarraystretch{1.7}\begin{tabular}{c|ccccc}
Class (cm)&$[150;160[$&$[160;170[$&$[170;180[$&$[180;190[$&$[190;200[$\\\hline
Count&8&22&35&28&7\\
Relative frequency (\%)&\mxblank{8}&\mxblank{22}&\mxblank{35}&\mxblank{28}&\mxblank{7}\\
Centre (cm)&\mxblank{155}&\mxblank{165}&\mxblank{175}&\mxblank{185}&\mxblank{195}\\
ACC&\mxblank{8}&\mxblank{30}&\mxblank{65}&\mxblank{93}&\mxblank{100}
\end{tabular}\end{center}
\mxanswerspace[4]{Width $10$ cm, total count $100$. Relative frequencies as percentages: $8,22,35,28,7$. Centres: $155,165,175,185,195$ cm. The weighted sum of the centres is $17540$, so the estimated mean is $175.4$ cm. The central ranks $50,51$ lie in $[170;180[$, the median class.}
\begin{example}If the exact class means were $154;166;174;184;196$ cm, what would the overall mean be?\end{example}
\mxanswerspace[2.4]{$(8\times154+22\times166+35\times174+28\times184+7\times196)/100=174.98$ cm.}
\mxsection{Histogram and cumulative frequencies}
\begin{keypoint}[Representations]
The area of each rectangle of a histogram is proportional to the count. For equal widths, the heights can be the counts. The cumulative frequency polygon joins the points (upper bound; cumulative frequency), starting at the first lower bound with $0$.
\end{keypoint}

\begin{method}[Estimating the median]
Under the assumption of uniformity in $[170;180[$, the cumulative frequency rises linearly from $30\,\%$ to $65\,\%$. Look for the abscissa corresponding to $50\,\%$.
\end{method}
\begin{example}Calculate this estimate and state its scope.\end{example}
\mxanswerspace[4]{Solve $(m-170)/10=(50-30)/(65-30)$, giving $m=170+10\times20/35\approx175.71$ cm. This is an estimate under an assumption of uniformity, not the exact median of the individual data. The class counts alone do not determine the latter.}
