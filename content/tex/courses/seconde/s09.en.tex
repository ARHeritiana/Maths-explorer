\section{Proportion and proportion of a proportion}
\begin{keypoint}[Definitions]
In a nonempty finite set $E$, the proportion of a subset $F$ is $p=\Card(F)/\Card(E)$; $0\leq p\leq1$. Its expression as a percentage is $100p\,\%$.
If $G\subseteq F\subseteq E$, $F\ne\varnothing$, and if $p_1$ is the proportion of $F$ in $E$ and $p_2$ that of $G$ in $F$, the proportion of $G$ in $E$ is $p_1p_2$.
\end{keypoint}
\begin{example}A school has $1500$ students, of whom $652$ are day boarders. Calculate their proportion. In a class, $\pc{60}$ of the students play sport; among them, $\pc{40}$ are boys. What proportion of the class do the boys who play sport represent?\end{example}
\mxanswerspace[4]{$652/1500\approx\num{0,4347}$, that is about $\pc{43,47}$. For the class, $\num{0,60}\times\num{0,40}=\num{0,24}$, that is $\pc{24}$. The second percentage has the sporty students as its reference, not the whole class.}
\begin{example}In a fictional school of $800$ students, $\pc{35}$ play a sport at a club; among them, $\pc{45}$ play a team sport. Calculate the two counts and the final proportion.\end{example}
\mxanswerspace[4]{$800\times\num{0,35}=280$ students at a club. $280\times\num{0,45}=126$ in a team sport. The final proportion is $126/800=\num{0,1575}=\pc{15,75}$.}
\begin{pitfall}Proportion and rate of change do not answer the same question: a proportion compares a part to a whole, a rate compares a change to an initial value.\end{pitfall}

\mxsection{Absolute change and rate of change}
\begin{keypoint}[Definition]
For an initial value $V_I>0$ and a final value $V_F>0$, the absolute change is $V_F-V_I$; the rate of change is
\[t=\frac{V_F-V_I}{V_I}.\]
The rate has no unit. It is positive for an increase, negative for a decrease. The reference is always the initial value.
\end{keypoint}
\begin{example}A cheese is sold for $\num{2,13}$ euros instead of $\num{3,55}$ euros. Calculate the absolute change and the rate.\end{example}
\mxanswerspace[3.2]{Absolute change: $\num{2,13}-\num{3,55}=-\num{1,42}$ euro. Rate: $-\num{1,42}/\num{3,55}=-\num{0,40}$, that is a decrease of $\pc{40}$.}
\begin{keypoint}[Multiplier]
$CM=V_F/V_I=1+t$, so $V_F=V_I(1+t)$. To recover $V_I$, divide $V_F$ by $1+t$ (nonzero).
\end{keypoint}
\begin{example}A fictional item priced $50000$ Ar increases by $\pc{12}$, then separately consider a $\pc{12}$ discount on the initial price. Give the final prices. A price after a $\pc{20}$ increase is $72000$ Ar; find its initial price.\end{example}
\mxanswerspace[4]{Increase: $50000\times\num{1,12}=56000$ Ar. Discount: $50000\times\num{0,88}=44000$ Ar. Last question: $V_I=72000/\num{1,20}=60000$ Ar.}
\begin{fluency}[5--10 min]Expand $(x-3)^2$; solve $1/x=5$; give the mean of $8;10;12$.\mxanswer{$x^2-6x+9$ ; $x=1/5$ ; $10$.}\end{fluency}

\mxsection{Successive changes}
\begin{keypoint}[Property]
Successive changes with rates $t_1,t_2$ have overall multiplier
\[CM_G=(1+t_1)(1+t_2),\qquad t_G=(1+t_1)(1+t_2)-1.\]
We multiply the multipliers; we do not simply add the rates.
\end{keypoint}
\begin{example}A manufacturer sold $5600$ bicycles in 2022, which is $\pc{25}$ more than in 2021. Its sales decrease by $\pc{15}$ in 2023. Determine the sales of 2021 and 2023, then the overall rate between these two years.\end{example}
\mxanswerspace[4]{2021: $5600/\num{1,25}=4480$ bicycles. 2023: $5600\times\num{0,85}=4760$ bicycles. Overall multiplier $\num{1,25}\times\num{0,85}=\num{1,0625}$, so an overall increase of $\pc{6,25}$. Check: $(4760-4480)/4480=\num{0,0625}$.}
\begin{example}A price increases by $\pc{10}$ then decreases by $\pc{10}$. Does it return to the initial price?\end{example}
\mxanswerspace[3.2]{$CM_G=\num{1,10}\times\num{0,90}=\num{0,99}$. The final price is $\pc{99}$ of the initial price: an overall decrease of $\pc{1}$. The two rates refer to different reference values.}
\begin{example}A rate goes from $\pc{20}$ to $\pc{25}$. Explain the difference between the increase in percentage points and the relative change.\end{example}
\mxanswerspace[3.2]{The gap is $5$ percentage points. The relative change is $(\num{0,25}-\num{0,20})/\num{0,20}=\num{0,25}$, that is an increase of $\pc{25}$.}

\mxsection{Reverse change}
\begin{keypoint}[Property]
To return to the initial value after a change of rate $t>-1$, use the multiplier $1/(1+t)$. The reverse rate is
\[t_{\text{reverse}}=\frac1{1+t}-1=\frac{-t}{1+t}.\]
\end{keypoint}
\begin{example}Which rate compensates a $\pc{25}$ increase? Which rate compensates a $\pc{20}$ decrease?\end{example}
\mxanswerspace[4]{After a $\pc{25}$ increase, the reverse multiplier is $1/\num{1,25}=\num{0,8}$: a $\pc{20}$ decrease. After a $\pc{20}$ decrease, it is $1/\num{0,8}=\num{1,25}$: a $\pc{25}$ increase.}
\begin{example}A price undergoes a $\pc{30}$ discount, then a $\pc{10}$ increase. What additional percentage increase restores the initial price?\end{example}
\mxanswerspace[5]{The current multiplier is $\num{0,70}\times\num{1,10}=\num{0,77}$. The required multiplier is $1/\num{0,77}=100/77$. The rate is $23/77\approx\num{0,2987}$, that is about $\pc{29,87}$. Check $\num{0,77}\times(100/77)=1$.}
\begin{teachernote}Have students state the base of the percentage at each step. The reverse-rate formula is justified by a product equal to $1$; memorisation alone is fragile.\end{teachernote}
